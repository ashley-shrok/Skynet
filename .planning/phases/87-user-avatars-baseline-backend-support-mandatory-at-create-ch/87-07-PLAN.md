    ```
    describe("Phase 87 end-to-end lifecycle", () => {
      it("create → serve → change → serve → delete → verify-clean", async () => {
        // Step 1: POST /users/create
        const createRes = await multipartRequestMixed(server, {
          path: "/users/create",
          textFields: { username: "alice", password: "s3cret" },
          file: { fieldName: "avatar", filename: "a.png", contentType: "image/png", bytes: validPngBytes },
        });
        expect(createRes.status).toBe(200);
        const aliceId = createRes.body.id;
        expect(aliceId).toBeTruthy();

        // Assert row + file
        const rowAfterCreate = /* SELECT username, avatar_path FROM users WHERE id = aliceId */;
        expect(rowAfterCreate.avatar_path).toBe(`${aliceId}.png`);
        const pngPathAfterCreate = path.join(USER_AVATARS_DIR, `${aliceId}.png`);
        const bytesOnDisk = await fs.readFile(pngPathAfterCreate);
        expect(bytesOnDisk.equals(validPngBytes)).toBe(true);

        // Assert save-trigger spy
        expect(saveSpy).toHaveBeenCalledWith("phase-87-user-avatar-create");

        // Step 2: login Alice, obtain JWT
        const jwt = await loginAndGetJwt(server, "alice", "s3cret");
        expect(jwt).toBeTruthy();

        // Step 3: GET /users/${aliceId}/avatar
        const getRes1 = await getWithAuth(server, `/users/${aliceId}/avatar`, jwt);
        expect(getRes1.status).toBe(200);
        expect(getRes1.headers["content-type"]).toBe("image/png");
        expect(Buffer.isBuffer(getRes1.body)).toBe(true);
        expect((getRes1.body as Buffer).equals(validPngBytes)).toBe(true);

        // Step 4: PUT /users/${aliceId}/avatar with WebP bytes (ext-swap case)
        saveSpy.mockClear();  // reset so next assertion is scoped to change endpoint
        const putRes = await putMultipartWithAuth(server, {
          path: `/users/${aliceId}/avatar`,
          fieldName: "avatar",
          filename: "a.webp",
          contentType: "image/webp",
          fileBytes: validWebpBytes,
        }, jwt);
        expect(putRes.status).toBe(200);
        expect(putRes.body.avatarPath).toBe(`${aliceId}.webp`);

        // Assert row updated
        const rowAfterChange = /* SELECT avatar_path FROM users WHERE id = aliceId */;
        expect(rowAfterChange.avatar_path).toBe(`${aliceId}.webp`);

        // Assert new file exists, old file unlinked
        const webpPath = path.join(USER_AVATARS_DIR, `${aliceId}.webp`);
        const newBytes = await fs.readFile(webpPath);
        expect(newBytes.equals(validWebpBytes)).toBe(true);
        await expect(fs.access(pngPathAfterCreate)).rejects.toThrow();  // old file gone

        // Assert save-trigger spy for change
        expect(saveSpy).toHaveBeenCalledWith("phase-87-user-avatar-change");

        // Step 5: GET /users/${aliceId}/avatar again — should return WebP
        const getRes2 = await getWithAuth(server, `/users/${aliceId}/avatar`, jwt);
        expect(getRes2.status).toBe(200);
        expect(getRes2.headers["content-type"]).toBe("image/webp");
        expect((getRes2.body as Buffer).equals(validWebpBytes)).toBe(true);

        // Step 6: DELETE /users/delete-account (self-delete)
        const delRes = await deleteWithAuth(server, "/users/delete-account", jwt);
        expect([200, 204]).toContain(delRes.status);  // match whichever the current endpoint returns

        // Step 7: verify row gone + file gone
        const rowAfterDelete = /* SELECT * FROM users WHERE id = aliceId */;
        expect(rowAfterDelete).toBeUndefined();  // or empty rows
        await expect(fs.access(webpPath)).rejects.toThrow();

        // Step 8: verify no stray files
        const remaining = await fs.readdir(USER_AVATARS_DIR).catch(() => []);
        expect(remaining).toEqual([]);
      });
    });
    ```
