```
import { CLIENT_BUILD_ID } from "./client-build-id";

/**
 * Phase 132 SKEW-04: stamped fetch wrapper. Enumerated raw-fetch call sites
 * (streaming, SSE, fire-and-forget beacons) go through this helper INSTEAD OF
 * the global `fetch` so the client-build header is stamped by construction.
 *
 * OUT OF SCOPE (do NOT rewrite to use stampedFetch): external URL fetches
 * (avatar candidates from `pravatar.cc` etc.) and Electron server-config
 * probes hitting user-configured remote installs. Those keep raw `fetch()`.
 */
export async function stampedFetch(
  input: RequestInfo | URL,
  init: RequestInit = {},
): Promise<Response> {
  const headers = new Headers(init.headers ?? {});
  headers.set("X-Skynet-Client-Build", CLIENT_BUILD_ID);
  return fetch(input, { ...init, headers });
}
```

Notes:

- The `.set()` call OVERWRITES any pre-existing `X-Skynet-Client-Build` — if a caller passed one manually, we win. Correct-by-construction.
- Returns the raw fetch Response — streaming callers still get `response.body.getReader()`.
