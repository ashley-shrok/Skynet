```
import { test, expect } from "@playwright/test";
import { loginViaUI, readCreds } from "./helpers/auth";

test.describe("Phase 132 SKEW-15: version-drift hard-lock end-to-end", () => {
  test("modal activates on server-tag mismatch and reload recovers", async ({ page }) => {
    await loginViaUI(page, readCreds());
    await expect(page).not.toHaveURL(/\/login/);

    // Interpose the next axios request to Skynet's own backend and inject a
    // mismatched X-Skynet-Server-Build header. This simulates a mid-session
    // deploy where the server jumped versions while the tab was open.
    await page.route("**/host/**", async (route) => {
      const response = await route.fetch();
      const headers = { ...response.headers(), "x-skynet-server-build": "totally-different-build-id" };
      await route.fulfill({ response, headers });
    });

    // Trigger any user action that fires an axios request. The dashboard
    // typically polls host status on load; if not, click a nav element that
    // does. Grep the app for the closest reliable-fires-an-axios-call button
    // if this doesn't fire naturally.
    await page.reload().catch(() => {}); // re-fetch host list = one axios call

    // Assert the skew-lock modal renders. Modal uses role="dialog" +
    // aria-labelledby="skynet-skew-lock-title" (see Plan 02 Task 3).
    const modal = page.getByRole("dialog", { name: /newer version/i });
    await expect(modal).toBeVisible({ timeout: 5000 });

    // Restore the route so the reload sees a fresh (non-injected) response.
    await page.unroute("**/host/**");

    // Click Reload — Playwright's page.reload is a real navigation, so the
    // Reload button's window.location.reload() will trigger a full reload.
    // BUT: sessionStorage counter accumulates; the sentinel activates after
    // 4 within 60s. This spec triggers once, so it's safely under threshold.
    await modal.getByRole("button", { name: /reload/i }).click();
    await page.waitForLoadState("networkidle");

    // After reload, modal should be gone (fresh page has no locked state).
    await expect(modal).not.toBeVisible();
  });
});
```

Verify the modal's accessible name matches `/newer version/i` — Plan 02 Task 3 sets the title to "A newer version is available" in normal mode.
