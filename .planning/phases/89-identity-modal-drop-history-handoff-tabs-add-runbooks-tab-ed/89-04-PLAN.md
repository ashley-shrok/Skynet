      setActiveTab(null);

      enumerateRunbookFiles(hostId, roleName, runbookName)

          // D-07 + SkillsEditorModal L171: auto-select first file on open

        .catch((err: unknown) => {
          if (!cancelled)

              status: "error",
              error: err instanceof Error ? err.message : "Failed to load files",
            });
        });
      return () => {
        cancelled = true;
      };
    }, [open, hostId, roleName, runbookName]);
