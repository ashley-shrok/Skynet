    ```typescript
    // Phase 79 Plan 08 — one-shot admin-gated migration.
    // Mints fresh Matrix tokens for every registered human (via loginAsUser),
    // writes /state/<humanName>.token 0600, then deletes any legacy
    // .cred files under TG_BRIDGE_STATE_DIR. Idempotent — safe to re-run.
    // Per CONTEXT § 4B + RESEARCH § A4/A6 (Laura has no mxid yet — skipped).
    router.post(
      "/migrate-cred-files",
      express.json(),
      requireAdmin,
      async (req: Request, res: Response): Promise<void> => {
        const adminUserId = (req as AuthenticatedRequest).userId;
        try {
          // Import lazily to avoid a circular-module problem at boot.
          const { db } = await import("../database/db/index.js");
          const { users } = await import("../database/db/schema.js");
          const rows = await db.select({ name: users.username, mxid: users.mxid }).from(users);

          const results: Array<{ humanName: string; mxid: string | null; status: string; error?: string }> = [];
          for (const row of rows) {
            // Blocker B-4 (verified at revision time): users.username is already
            // lowercase for Ashley/Zoey. The .toLowerCase() below is a defensive
            // no-op — kept in case future data drifts.
            const humanName = String(row.name ?? "").toLowerCase();
            if (!humanName) continue;
            if (!row.mxid) {
              results.push({ humanName, mxid: null, status: "skipped-no-mxid" });
              continue;
            }
            const mintResult = await mintAndWriteHumanToken(row.mxid, humanName);
            if (mintResult.ok) {
              results.push({ humanName, mxid: row.mxid, status: "minted" });
            } else {
              results.push({ humanName, mxid: row.mxid, status: "failed", error: mintResult.error });
            }
          }

          // Delete any legacy .cred files under the shared volume.
          const entries = await fsp.readdir(TG_BRIDGE_STATE_DIR).catch(() => [] as string[]);
          const credFiles = entries.filter((e) => e.endsWith(".cred"));
          const deletedCredFiles: string[] = [];
          for (const f of credFiles) {
            try {
              await fsp.unlink(`${TG_BRIDGE_STATE_DIR}/${f}`);
              deletedCredFiles.push(f);
            } catch (err) {
              authLogger.warn("migrate-cred-files: failed to unlink cred file", {
                operation: "matrix_admin_migrate_cred_files_unlink_failed",
                filename: f,
                error: err instanceof Error ? err.message : "unknown",
              });
            }
          }

          authLogger.info("matrix-admin migrate-cred-files completed", {
            operation: "matrix_admin_migrate_cred_files",
            adminId: adminUserId,
            mintedCount: results.filter((r) => r.status === "minted").length,
            failedCount: results.filter((r) => r.status === "failed").length,
            skippedCount: results.filter((r) => r.status === "skipped-no-mxid").length,
            deletedCredFilesCount: deletedCredFiles.length,
          });

          res.json({ ok: true, results, deletedCredFiles });
        } catch (err) {
          authLogger.error("Failed to migrate cred files", err);
          res.status(500).json({ error: "Failed to migrate cred files" });
        }
      },
    );
    ```
