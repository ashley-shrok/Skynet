    ```typescript
    // Phase 79 Plan 07 — Telegram bridge frontend API client.
    // Thin wrappers around authApi for the 4 routes defined in Plan 03:
    //   POST /telegram/validate      — proxied getMe check
    //   POST /telegram/activate      — write telegram_bot_tokens row + register bot
    //   POST /telegram/disconnect    — delete row + unregister bot
    //   GET  /telegram/:identityKey  — status view for the tab
    //
    // Bot token is only ever placed in POST body — never URL query, never logged.

    import { authApi, handleApiError } from "@/main-axios";

    export interface ValidateResp {
      ok: boolean;
      botUsername?: string;
      botId?: number;
      firstName?: string;
      error?: string;
    }

    export async function postTelegramValidate(input: { botToken: string }): Promise<
      | { ok: true; botUsername: string; botId: number; firstName: string }
      | { ok: false; error: string }
    > {
      try {
        const resp = await authApi.post<ValidateResp>("/telegram/validate", {
          botToken: input.botToken,
        });
        const data = resp.data;
        if (data.ok && data.botUsername && typeof data.botId === "number") {
          return { ok: true, botUsername: data.botUsername, botId: data.botId, firstName: data.firstName ?? "" };
        }
        return { ok: false, error: data.error ?? "Telegram rejected token" };
      } catch (err) {
        return { ok: false, error: handleApiError(err) };
      }
    }

    export async function postTelegramActivate(input: {
      identityKey: string;
      botToken: string;
      humanUserId: string;
      telegramChatId?: string | null;
    }): Promise<{ ok: true; botUsername: string } | { ok: false; error: string; status?: number }> {
      try {
        const resp = await authApi.post<{ ok: boolean; botUsername?: string; error?: string }>(
          "/telegram/activate",
          input,
        );
        if (resp.data.ok && resp.data.botUsername) {
          return { ok: true, botUsername: resp.data.botUsername };
        }
        return { ok: false, error: resp.data.error ?? "activation failed" };
      } catch (err: any) {
        return { ok: false, error: handleApiError(err), status: err?.response?.status };
      }
    }

    export async function postTelegramDisconnect(input: {
      identityKey: string;
    }): Promise<{ ok: true } | { ok: false; error: string; status?: number }> {
      try {
        const resp = await authApi.post<{ ok: boolean; error?: string }>("/telegram/disconnect", input);
        if (resp.data.ok) return { ok: true };
        return { ok: false, error: resp.data.error ?? "disconnect failed" };
      } catch (err: any) {
        return { ok: false, error: handleApiError(err), status: err?.response?.status };
      }
    }

    export interface StatusResp {
      status: "unconfigured" | "connected";
      botUsername?: string;
      telegramChatId?: string | null;
    }

    export async function getTelegramStatus(
      identityKey: string,
    ): Promise<
      | { status: "unconfigured" }
      | { status: "connected"; botUsername: string; telegramChatId: string | null }
      | { status: "error"; error: string }
    > {
      try {
        const resp = await authApi.get<StatusResp>(`/telegram/${encodeURIComponent(identityKey)}`);
        if (resp.data.status === "connected" && resp.data.botUsername) {
          return {
            status: "connected",
            botUsername: resp.data.botUsername,
            telegramChatId: resp.data.telegramChatId ?? null,
          };
        }
        return { status: "unconfigured" };
      } catch (err) {
        return { status: "error", error: handleApiError(err) };
      }
    }
    ```
