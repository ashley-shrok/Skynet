        # is_coordinator <identity_file_path>
        # Returns 0 IFF the file has `coordinator: true` as a top-level key on its own line

        is_coordinator() {
          local identity_file="$1"

        get_freshness_epoch() {

          fi
          printf '%s' "${mtime:-0}"
