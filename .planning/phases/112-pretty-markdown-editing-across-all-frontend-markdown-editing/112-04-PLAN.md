    Shape (rough — executor derives exact code):

    ```
    // MarkdownEditor.frontmatter-fixture.tsx (self-contained, dev-only, Playwright-loaded)
    import { StrictMode, useEffect, useState } from "react";
    import { createRoot } from "react-dom/client";
    import { MarkdownEditor } from "./MarkdownEditor";

    const SAMPLES: Record<string, string> = {
      identity: "---\nrole: box-maintainer\ndisplayName: Cedar\ntask: something\n---\n\n# Hello\n\nbody here\n",
      malformed: "---\nrole: box-maintainer\ndisplayName: Cedar\ntask: Fix: bug\n---\n\n# Hello\n\nbody here\n",
      default: "---\nkey: value\n---\n\n# Default\n",
    };

    function pickSample(): string {
      // Prefer the URL query param on the script's own module URL (import.meta.url includes ?sample=identity),
      // fall back to window.location.search for the case where the test uses page.goto with the query.
      const scriptUrl = new URL(import.meta.url);
      const sampleFromScript = scriptUrl.searchParams.get("sample");
      const sampleFromWindow = new URLSearchParams(window.location.search).get("sample");
      const key = sampleFromScript ?? sampleFromWindow ?? "identity";
      return SAMPLES[key] ?? SAMPLES.identity;
    }

    function Fixture(): JSX.Element {
      const [content, setContent] = useState(pickSample);
      useEffect(() => {
        (window as any).__editorGetMarkdown = () => content;
        return () => { delete (window as any).__editorGetMarkdown; };
      }, [content]);
      return <MarkdownEditor filename="fixture.md" content={content} onChange={setContent} />;
    }

    if (import.meta.env.DEV && !document.getElementById("__mdx_fixture_root")) {
      // Optional: hide the main app so the fixture is the only thing on screen.
      const appRoot = document.getElementById("root");
      if (appRoot) appRoot.style.display = "none";
      const host = document.createElement("div");
      host.id = "__mdx_fixture_root";
      host.style.cssText = "position:fixed;inset:0;background:#0b0e14;z-index:9999;padding:24px;overflow:auto;";
      document.body.appendChild(host);
      createRoot(host).render(<StrictMode><Fixture /></StrictMode>);
    }
    ```
