<objective>
Prove the composed system works. Every prior plan in Phase 75 tested its own piece in isolation with mocked collaborators. This plan wires the pieces together (with mocks pushed OUT to just the DB/SSH/filesystem boundary) and asserts end-to-end behavior of the load-bearing user-visible properties from D-17 and D-18:

- Startup pass runs at container start (well, at test-suite setup — same shape), enumerates the right hosts, sweeps each one serially, marks successes done, leaves failures for retry.
- On-add trigger from the POST /host/db/host route handler fires a real sweep on the real orchestrator via the real singleton, and the HTTP response does not block.
- Retry loop picks up failures at the 30s tick and does NOT re-sweep successes.
- Persistent-failure alert fires exactly once when a host fails N=3 times in a row.

Anything a real Skynet startup would exercise for the substrate-sweep path is exercised here — except for actual SSH connections and actual filesystem reads of bundled substrate bytes, which are mocked at the outermost boundary.

Purpose: Deliver the fully-composed D-17 and D-18 integration tests. Catch composition bugs that unit tests miss (bad relative import paths, wrong singleton scope, wrong deps injection shape, wrong microtask draining semantics under fake timers, etc.).

Output: One test file that stress-tests the composition end-to-end.
</objective>

      beforeEach(() => {
        vi.useFakeTimers();
        vi.clearAllMocks();
        __resetSubstrateOrchestrator(); // from 75-05 Task 1
      });
      afterEach(() => {
        vi.useRealTimers();

      describe("I-STARTUP (D-17): startup pass end-to-end", () => {
        it("walks all fixture substrate hosts serially and marks each done on clean sweep", async () => {
          // Set up stubbed getDb returning 3 rows for listSubstrateHosts
          const stubDb = makeStubDb({
            substrateHosts: [
              { id: 1, name: "host1", credentialId: 100, ip: "10.0.0.1", port: 22, username: "root", ... },
              { id: 2, name: "host2", credentialId: 101, ... },
              { id: 3, name: "host3", credentialId: 102, ... },

            credentials: [
              { id: 100, systemPassword: "ct1", systemKey: null, ... },
              { id: 101, systemPassword: null, systemKey: "ct2", ... },
              { id: 102, systemPassword: "ct3", ... },
            ],
          });

          // Real acquireChannel that returns a mock SshChannel with canned exec responses
          // that make FLEET_SUBSTRATE_CATALOG per-item idempotence return "up-to-date" for all items
          const sweepOrder: string[] = [];
          const acquireChannel = vi.fn(async (host) => ({
            exec: vi.fn(async (cmd) => {
              sweepOrder.push(host.id);
              return "..."; // canned response that satisfies catalog items
            }),
          }));

          const orch = createServerSubstrateOrchestrator({
            listSubstrateHosts: () => listSubstrateHosts({ getDb: () => stubDb }),
            acquireChannel,
            releaseChannel: vi.fn(),
            setInterval,
            clearInterval,
            now: () => Date.now(),
            retryIntervalMs: 30000,
            persistentFailureThreshold: 3,
          });

          await orch.start();
          await Promise.resolve(); // drain microtasks

          // Assert: all 3 hosts were swept
          expect(acquireChannel).toHaveBeenCalledTimes(3);
          // Assert: serial order (first host id acquired before second's exec fires, etc.)
          expect(sweepOrder.slice(0, 3)).toEqual(["1", "2", "3"]);
          // Assert: sweepTickCount incremented for the startup pass
          expect(orch.getSweepTickCount()).toBe(1);

          orch.stop();
        });
      });

      describe("I-RETRY (D-17): retry loop picks up failures, skips successes", () => {
        it("30s tick re-sweeps failed host, skips already-succeeded host", async () => {
          // 2 hosts. First always succeeds; second's acquireChannel returns null (failure).
          const stubDb = makeStubDb({
            substrateHosts: [
              { id: 1, name: "ok-host", credentialId: 100, ... },
              { id: 2, name: "fail-host", credentialId: 101, ... },
            ],
            credentials: [ ... ],
          });
          const acquireCallsPerHost = new Map<string, number>();
          const acquireChannel = vi.fn(async (host) => {
            acquireCallsPerHost.set(host.id, (acquireCallsPerHost.get(host.id) ?? 0) + 1);
            if (host.id === "2") return null; // simulate unreachable
            return { exec: vi.fn(async () => "...") };
          });

          const orch = createServerSubstrateOrchestrator({ ...normal deps..., acquireChannel });
          await orch.start();
          await Promise.resolve();

          // After startup: host 1 succeeded (marked done), host 2 failed (not marked)
          expect(acquireCallsPerHost.get("1")).toBe(1);
          expect(acquireCallsPerHost.get("2")).toBe(1);
          expect(orch.getSweepTickCount()).toBe(1);

          // Advance 30s → retry tick fires
          await vi.advanceTimersByTimeAsync(30000);
          await Promise.resolve();

          // host 1 stays at 1 acquire call (already done), host 2 gets another attempt
          expect(acquireCallsPerHost.get("1")).toBe(1);
          expect(acquireCallsPerHost.get("2")).toBe(2);
          expect(orch.getSweepTickCount()).toBe(2);

          orch.stop();
        });
      });

      describe("I-PERSISTENT (D-06): loud alert fires exactly once at N failures", () => {
        it("3 consecutive failures → logPersistentFailure fires once", async () => {
          const stubDb = makeStubDb({ substrateHosts: [{id:1,name:"h1",...}], credentials: [...] });
          const acquireChannel = vi.fn(async () => null); // always fail

          const orch = createServerSubstrateOrchestrator({
            ...normal deps..., acquireChannel,
            persistentFailureThreshold: 3,
          });

          await orch.start();
          await Promise.resolve();
          // Failure 1 recorded

          await vi.advanceTimersByTimeAsync(30000);
          await Promise.resolve();
          // Failure 2 recorded

          await vi.advanceTimersByTimeAsync(30000);
          await Promise.resolve();
          // Failure 3 recorded → logPersistentFailure fires

          // Assert: logger.warn was called with operation "fleet_substrate_host_persistent_failure" exactly once
          const persistentCalls = (systemLogger.warn as any).mock.calls.filter(
            (call: any[]) => call[1]?.operation === "fleet_substrate_host_persistent_failure",
          );
          expect(persistentCalls.length).toBe(1);
          expect(persistentCalls[0][1].consecutiveFailures).toBe(3);
          expect(persistentCalls[0][1].hostName).toBe("h1");

          // 4th failure does NOT re-alert
          await vi.advanceTimersByTimeAsync(30000);
          await Promise.resolve();
          const persistentCallsAfter = (systemLogger.warn as any).mock.calls.filter(
            (call: any[]) => call[1]?.operation === "fleet_substrate_host_persistent_failure",
          );
          expect(persistentCallsAfter.length).toBe(1); // still 1, not 2

          orch.stop();
        });
      });

      describe("I-ON-ADD (D-18): POST route → singleton → real orchestrator", () => {
        it("POST /host/db/host with substrate host fires sweep via real singleton, response does not block", async () => {
          // Set up a real orchestrator, register it in the singleton
          const acquireChannel = vi.fn(async () => new Promise((resolve) => setTimeout(() => resolve({exec: vi.fn(async () => "...")}), 500)));
          const orch = createServerSubstrateOrchestrator({
            listSubstrateHosts: () => listSubstrateHosts({ getDb: () => stubDb }),
            acquireChannel,
            releaseChannel: vi.fn(),
            setInterval,
            clearInterval,
            now: () => Date.now(),
            retryIntervalMs: 30000,

          const responseTime = Date.now() - startTime;

          expect(response.status).toBe(200);
          expect(responseTime).toBeLessThan(100); // response fires FAST, sweep is slow (500ms)

          // Give the microtask a moment to fire but not to complete
          await Promise.resolve();
          expect(acquireChannel).toHaveBeenCalledTimes(1);
          expect(acquireChannel.mock.calls[0][0].id).toBe("<the new host id>");

          orch.stop();
        });
      });
    });
