Delete the RO gymnastics, the <20px delta heuristic that filters virtualizer writes from user scrolls, the sticky-vs-programmatic tracking, the rAF chain, the MutationObserver for per-child RO tracking, and the tall-bubble jump-to-different-area protection.       const scrollRef = useCallback((el: HTMLElement | null) => setScrollEl(el), []);
      const [isPinnedToBottom, setIsPinnedToBottom] = useState<boolean>(true);
      const pinnedRef = useRef<boolean>(true);

      useEffect(() => {
        if (!scrollEl) return;
        const onScroll = () => {
          const dist = scrollEl.scrollHeight - scrollEl.scrollTop - scrollEl.clientHeight;
          const pinned = dist <= BOTTOM_EPSILON;
          pinnedRef.current = pinned;
          setIsPinnedToBottom(pinned);
        };
        onScroll();
        scrollEl.addEventListener("scroll", onScroll, { passive: true });
        return () => scrollEl.removeEventListener("scroll", onScroll);
      }, [scrollEl]);

      useEffect(() => {
        if (!scrollEl) return;
        if (!pinnedRef.current) return;
        scrollEl.scrollTop = scrollEl.scrollHeight;
      }, [scrollEl, messageCount]);

      const scrollToBottomAndFollow = useCallback(() => {
        if (!scrollEl) return;
        scrollEl.scrollTop = scrollEl.scrollHeight;
        pinnedRef.current = true;
        setIsPinnedToBottom(true);
      }, [scrollEl]);

      return { scrollRef, scrollToBottomAndFollow, isPinnedToBottom };
