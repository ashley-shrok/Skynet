    (c) Add a click handler function inside the component:
    ```
    const handleArchiveClick = (roleName: string, roleDisplayLabel: string) => {
      // D-04: compute cascade preview frontend-side from useIdentities().
      // D-07 rehashing this at supervisor scan time is the supervisor's job;
      // D-11 handles the click-vs-scan enumeration race — we don't guard against it here.
      const affected = identities.filter(
        (i) => i.role === roleName && i.hostId === selectedHostId,
      );

      // D-03 first confirm: blast-radius disclosure. Complete list, no truncation.
      // Line-per-identity uses `task || displayName` fallback matching AppShell.tsx:894.
      const dialog1 =
        affected.length === 0
          ? `archive role ${roleDisplayLabel}? no identities hold it.`
          : `archive role ${roleDisplayLabel}? this will also archive ${affected.length} identities holding it:\n` +
            affected.map((i) => `• ${i.task || i.displayName}`).join("\n");
      if (!window.confirm(dialog1)) return;

      // D-03 second confirm: sanity tap. Byte-identical to identity-archive's copy.
      if (!window.confirm("are you sure? this can't be undone.")) return;

      // D-01 fire-and-forget: cascade complexity lives in the supervisor; the UI
      // does not wait. .catch() logs a structured warn matching IdentitySessionPane.tsx:232-239.
      if (selectedHostId != null) {
        void archiveRole(selectedHostId, roleName).catch((err) => {
          console.warn({
            operation: "role_archive_failed",
            hostId: selectedHostId,
            roleName,
            errMessage: err instanceof Error ? err.message : String(err),
          });
        });
      }
    };
    ```
